SER_remove_ai_army_and_factories={
	SER={
		delete_unit={disband=no} #Equipment is not kept
		every_owned_state={
			limit={
				is_core_of=SER
				is_owned_by=SER
			}
			set_variable={civs=industrial_complex_level}
			set_variable={mils=arms_factory_level}
			remove_building={ type=industrial_complex level=civs }
			remove_building={ type=arms_factory level=mils }
		}
	}
}

SER_conquer_ai={
	AUS={
		annex_country={
			target=SER
			transfer_troops=no
		}
	}
	150={
		remove_dynamic_modifier = { modifier = FFU_no_ports_modifier }
	}
}
