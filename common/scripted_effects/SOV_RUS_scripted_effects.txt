SOV_RUS_check_and_manage_losing_cores_for_bop_effect = {

	#SEARCH KEYS:
	# LITTLE BOP PROGRESS TOWARDS BAD SIDE FOR LOSING POLISH SOV CORES
	# LITTLE BOP PROGRESS TOWARDS BAD SIDE FOR LOSING BELARUSIAN SOV CORES
	# MODERATE BOP PROGRESS TOWARDS BAD SIDE FOR LOSING RUSSIAN SOV CORES
	# SEVERE BOP PROGRESS TOWARDS GOOD SIDE FOR HOLDING AUS CORES
	# SEVERE BOP PROGRESS TOWARDS GOOD SIDE FOR HOLDING GER CORES
	# SEVERE BOP PROGRESS TOWARDS GOOD SIDE FOR HOLDING TUR CORES



	# SEVERE BOP PROGRESS TOWARDS GOOD SIDE FOR HOLDING AUS CORES
	if = {
		limit = {
			any_state = {
				is_core_of = AUS
				is_controlled_by = ROOT
			}
		}
		add_power_balance_modifier = {
			id = SOV_RUS_balance_of_powers
			modifier = SOV_RUS_bop_austrias_as_warriors
		}
	}
	else_if = {
		limit = {
			has_power_balance_modifier = {
				id = SOV_RUS_balance_of_powers
				modifier = SOV_RUS_bop_austrias_as_warriors
			}
		}
		remove_power_balance_modifier = {
			id = SOV_RUS_balance_of_powers
			modifier = SOV_RUS_bop_austrias_as_warriors
		}
	}

	# SEVERE BOP PROGRESS TOWARDS GOOD SIDE FOR HOLDING GER CORES
	if = {
		limit = {
			any_state = {
				is_core_of = GER
				is_controlled_by = ROOT
			}
		}
		add_power_balance_modifier = {
			id = SOV_RUS_balance_of_powers
			modifier = SOV_RUS_bop_breaking_the_spike
		}
	}
	else_if = {
		limit = {
			has_power_balance_modifier = {
				id = SOV_RUS_balance_of_powers
				modifier = SOV_RUS_bop_breaking_the_spike
			}
		}
		remove_power_balance_modifier = {
			id = SOV_RUS_balance_of_powers
			modifier = SOV_RUS_bop_breaking_the_spike
		}
	}

	# SEVERE BOP PROGRESS TOWARDS GOOD SIDE FOR HOLDING TUR CORES
	if = {
		limit = {
			any_state = {
				is_core_of = TUR
				is_controlled_by = ROOT
			}
		}
		add_power_balance_modifier = {
			id = SOV_RUS_balance_of_powers
			modifier = SOV_RUS_bop_another_russo_turkish
		}
	}
	else_if = {
		limit = {
			has_power_balance_modifier = {
				id = SOV_RUS_balance_of_powers
				modifier = SOV_RUS_bop_another_russo_turkish
			}
		}
		remove_power_balance_modifier = {
			id = SOV_RUS_balance_of_powers
			modifier = SOV_RUS_bop_another_russo_turkish
		}
	}
}
