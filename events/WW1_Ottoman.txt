##Event: Black Sea Raid
country_event = {
	id = ffuottoman.7
	title = ffuottoman.7.title
	desc = ffuottoman.7.desc
	picture = GFX_report_event_usa_heavy_cruiser
	fire_only_once = yes
	is_triggered_only = yes
	option = {
		name = ffuottoman.7.a
		ai_chance = { factor = 1}
		TUR = {
			set_variable = {
				Global.super_event=6
			}
			FFU_show_super_event = yes
		}
		hidden_effect = {
			every_country = {
				limit = {check_variable= { super_events_enabled > 0} }
				scoped_sound_effect="FFU_black_sea_raid" 
			}
		}
	}
	immediate = {
		GER = {
			add_to_faction = TUR
			if = {
				limit = { LBA = { exists=yes is_ai=no } }
				add_to_faction = LBA
			}
		}
		TUR = {
			declare_war_on={
				type=annex_everything
				target=ENG
			}
			declare_war_on={
				type=annex_everything
				target=FRA
			}
			if = {
				limit = {
					SOV = { has_war_with = GER }
				}
				declare_war_on = {
					type = annex_everything
					target = SOV
				}
			}
			if = {
				limit = {
					SER = { has_war_with = GER }
				}
				declare_war_on = {
					type = annex_everything
					target = SER
				}
			}
			if = {
				limit = {
					BEL = { exists = yes}
				}
				declare_war_on = {
					type = annex_everything
					target = BEL
				}
			}
		}
		if = {
			limit = { LBA = { exists = yes } }
			LBA = {
				declare_war_on={
					type=annex_everything
					target=ENG
				}
				declare_war_on={
					type=annex_everything
					target=FRA
				}
				if = {
					limit = {
						SOV = { has_war_with = GER }
					}
					declare_war_on = {
						type = annex_everything
						target = SOV
					}
				}
				if = {
					limit = {
						SER = { has_war_with = GER }
					}
					declare_war_on = {
						type = annex_everything
						target = SER
					}
				}
				if = {
					limit = {
						BEL = { exists = yes}
					}
					declare_war_on = {
						type = annex_everything
						target = BEL
					}
				}
			}
		}
	}
}
