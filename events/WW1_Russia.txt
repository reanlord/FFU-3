news_event = {
	id = ww1_russia.4
	title = ww1_russia.4.t
	desc = ww1_russia.4.d
	picture = EVENT_hoi4tgw_october_revolution
	#major = yes
	fire_only_once=yes
	mean_time_to_happen = {
		days = 0
	}

	trigger = {
		NOT = { has_global_flag = soviets_civil_war }
		SOV = { exists = yes }
		OR = {
			SOV = {
				is_power_balance_in_range = {
					id = SOV_RUS_balance_of_powers
					range = SOV_RUS_cap_range
				}
			}
			SOV = {
				NOT = { controls_state = 115 }
				NOT = {	controls_state = 323 }
				NOT = {	controls_state = 331 }
				NOT = {	controls_state = 172 }
			}
		}
	}

	immediate = {
		hidden_effect = {  
            every_country = {limit = {check_variable= { super_events_enabled > 0} }
				scoped_sound_effect="FFU_russian_civil_war"
			}
            set_global_flag = soviets_civil_war			
			RUS = {
				create_faction = "Comintern"
				
				set_politics = {
				    ruling_party = communism
				    last_election = "1912.9.1"
				    election_frequency = 60 
				    elections_allowed = no
				}
				set_popularities = { 
					neutrality = 0
					democratic = 10 
					fascism = 0   
					communism = 90 
				}
				add_ideas={
					ai_limiter
					RUS_civil_war
				}
				transfer_state=126	#Petrograd
			}
			SOV = {
				set_variable = {
					Global.super_event=3
				}
				FFU_show_super_event = yes
				set_country_flag = dual_power
				set_country_flag = kis_tsarnicholasabdicated
				set_country_flag = bolsheviks
				SOV_russia_abdication=yes
				country_event = { id = Troops_Home.2 days = 1 }
				country_event = { id = ww1_russia.7 days = 2 }
			}
		}
		SOV_central_powers_debuffs = yes
	}

	option = {
		name = ww1_russia.4.a
		trigger = { tag = RUS NOT = { tag = SOV } }
		ai_chance = { factor = 100 }
	}
	option = {
		name = ww1_russia.4.b
		trigger = { tag = RUS NOT = { tag = SOV } }
		ai_chance = { factor = 100 }
	}
	option = {
		name = ww1_russia.4.c
		trigger = { NOT = { tag = SOV tag = RUS } }
		ai_chance = { factor = 100 }
	}
}
