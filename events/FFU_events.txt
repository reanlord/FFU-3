country_event = {
	id = ffu.1
	title = ffu.1.t
	desc = ffu.1.d

	picture = GFX_report_event_tanks_mexico_city
	is_triggered_only = yes

	#hidden = yes

	option = {
		name = ffu.1.a

		every_country = {
			every_army_leader = {
				limit = {
					NOT = { has_trait = cheater }
					OR = {
						AND = {
							has_trait = panzer_expert
							has_trait = combined_arms_expert
						}
						AND = {
							has_trait = combined_arms_expert
							has_trait = cavalry_expert
						}
						AND = {
							has_trait = panzer_expert
							has_trait = cavalry_expert
						}
						AND = {
							has_trait = infantry_expert
							has_trait = ambusher
						}
						AND = {
							has_trait = invader_ii
							has_trait = naval_liason
						}
						AND = {
							has_trait = offensive_doctrine
							has_trait = defensive_doctrine
						}
						AND = {
							has_trait = organisational_leader
							has_trait = inspirational_leader
						}
					}
				}
				add_unit_leader_trait = cheater
			}
		}
	}
}

##Event: 1914 Chirstmas Truce
news_event = {
	id = ffu.3
	title = ffu.3.title
	desc = ffu.3.desc
	picture = GFX_1914_chirstmas_truce-122744
	is_triggered_only = yes
	major = yes
	fire_only_once = yes
	option = {
		name = ffu.3.a
		ai_chance = { factor = 1}
		FRA = {
			set_variable = {
				Global.super_event=8				
			}
			FFU_show_super_event = yes
		}
		hidden_effect = {
			every_country = {
				if = {
					limit = {check_variable= { super_events_enabled > 0} }
					scoped_sound_effect="FFU_christmas_truce"
				}
			}
		}
		hidden_effect = {
			every_country = {
				add_timed_idea = { idea = chistmas_truce  days = 14 }
			}
		}
	}
}
